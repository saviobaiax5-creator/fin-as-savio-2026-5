    let grupoAtivo = "Todos";
    let bancoSelecionadoId = "99pay";

    function definirFiltro(grupo, botao) {
      grupoAtivo = grupo;
      document.querySelectorAll(".filtro-btn").forEach(b => b.classList.remove("ativo"));
      botao.classList.add("ativo");
      atualizarLista();
    }

    function atualizarLista() {
      const termo = document.getElementById("input-busca").value.toLowerCase();
      const container = document.getElementById("container-bancos");

      const filtrados = INSTITUICOES.filter(item => {
        const bateGrupo = grupoAtivo === "Todos" || item.grupo === grupoAtivo;
        const bateTexto = item.nome.toLowerCase().includes(termo) || item.codigo.includes(termo);
        return bateGrupo && bateTexto;
      });

      container.innerHTML = filtrados.map(b => `
        <button type="button" class="card-banco ${b.id === bancoSelecionadoId ? 'selecionado' : ''}" onclick="selecionarBanco('${b.id}')">
