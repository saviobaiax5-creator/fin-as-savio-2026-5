### 3.2. Importador Excel (`src/services/salaryExcelImport.ts`)
```ts
import * as XLSX from "xlsx";

import type {
  SalaryCategory,
  SalarySource,
  SalaryRecord,
  SalaryImportResult,
} from "../types/salary";

const MONTHS = [
  "janeiro",
  "fevereiro",
  "março",
  "abril",
  "maio",
  "junho",
  "julho",
  "agosto",
  "setembro",
  "outubro",
  "novembro",
  "dezembro",
];

function createId(): string {
  return crypto.randomUUID();
}

function classifySource(name: string): SalaryCategory {
  const value = name.toLowerCase();

  if (value.includes("empréstimo")) return "loan";

  if (value.includes("13º") || value.includes("decimo")) {
    return "thirteenth";
  }

  if (value.includes("férias") || value.includes("ferias")) {
    return "vacation";
  }

  if (value.includes("fgts")) return "fgts";

  if (
    value.includes("vale") ||
    value.includes("pis") ||
    value.includes("abono")
  ) {
    return "benefit";
  }

  if (
    value.includes("unip") ||
    value.includes("unigrande") ||
    value.includes("cei")
  ) {
    return "teaching";
  }

  if (
    value.includes("extra") ||
    value.includes("pré projeto") ||
    value.includes("pre projeto")
  ) {
    return "occasional";
  }

  if (
    value.includes("best pix") ||
    value.includes("recarga") ||
    value.includes("celcoin")
  ) {
    return "credit";
  }

  if (value.includes("câmara") || value.includes("camara")) {
    return "salary";
  }

  return "other";
}

function normalizeAmount(value: unknown): number | null {
  if (value === null || value === undefined || value === "") {
    return null;
  }

  if (typeof value === "number") {
    return Number.isFinite(value) ? value : null;
  }

  if (typeof value === "string") {
    const normalized = value
      .replace("R$", "")
      .replace(/\./g, "")
      .replace(",", ".")
      .trim();

    const parsed = Number(normalized);

    return Number.isFinite(parsed) ? parsed : null;
  }

  return null;
}

export async function importSalaryExcel(
  file: File
): Promise<SalaryImportResult> {
  const warnings: string[] = [];
  const errors: string[] = [];

  const buffer = await file.arrayBuffer();

  const workbook = XLSX.read(buffer, {
    type: "array",
    cellDates: true,
    cellNF: true,
    cellText: true,
  });

  const sheet = workbook.Sheets["Planilha1"];

  if (!sheet) {
    throw new Error("A aba Planilha1 não foi encontrada.");
  }

  const rows = XLSX.utils.sheet_to_json(sheet, {
    header: 1,
    defval: null,
    raw: true,
  }) as unknown[][];

  const header = rows[0];

  if (!header || Number(header[1]) !== 2026) {
    throw new Error(
      "Não foi possível identificar o ano na tabela salarial."
    );
  }

  for (let i = 0; i < 12; i++) {
    const monthName = String(header[i + 2] ?? "")
      .trim()
      .toLowerCase();

    if (monthName !== MONTHS[i]) {
      throw new Error(
        `Cabeçalho incorreto na coluna ${i + 3}: esperado ${MONTHS[i]}.`
      );
    }
  }

  const year = Number(header[1]);

  const sources: SalarySource[] = [];
  const records: SalaryRecord[] = [];

  // Linhas 2 até 24 do Excel -> Índices 1 até 23 no array.
  for (let rowIndex = 1; rowIndex <= 23; rowIndex++) {
    const row = rows[rowIndex];

    if (!row) continue;

    const sourceName = String(row[0] ?? "").trim();

    if (!sourceName) continue;

    const source: SalarySource = {
      id: createId(),
      name: sourceName,
      category: classifySource(sourceName),
      active: true,
      createdAt: new Date().toISOString(),
    };

    sources.push(source);

    // Colunas C:N -> 12 meses.
    for (let monthIndex = 0; monthIndex < 12; monthIndex++) {
      const originalValue = row[monthIndex + 2];
      const amount = normalizeAmount(originalValue);

      if (amount === null) continue;

      if (amount < 0) {
        warnings.push(
          `Valor negativo: ${sourceName}, ${MONTHS[monthIndex]}.`
        );
      }

      records.push({
        id: createId(),
        sourceId: source.id,
        year,
        month: monthIndex + 1,
        amount,
        status: "unknown",
        origin: "excel",
        sourceRow: rowIndex + 1,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });
    }
  }

  if (sources.length === 0) {
    errors.push("Nenhuma fonte de renda foi encontrada.");
  }

  return {
    year,
    sources,
    records,
    warnings,
    errors,
  };
}

### 3.3. Cálculos (`src/utils/salaryCalculations.ts`)
```ts
import type { SalaryRecord, SalarySource } from "../types/salary";

export function getMonthlyTotal(
  records: SalaryRecord[],
  year: number,
  month: number
): number {
  return records
    .filter(
      (record) =>
        record.year === year &&
        record.month === month &&
        record.status === "received"
    )
    .reduce((total, record) => total + (record.amount ?? 0), 0);
}

export function getAnnualTotal(records: SalaryRecord[], year: number): number {
  return records
    .filter(
      (record) => record.year === year && record.status === "received"
    )
    .reduce((total, record) => total + (record.amount ?? 0), 0);
}

export function getSourceAnnualTotal(
  records: SalaryRecord[],
  sourceId: string,
  year: number
): number {
  return records
    .filter(
      (record) =>
        record.sourceId === sourceId &&
        record.year === year &&
        record.status === "received"
    )
    .reduce((total, record) => total + (record.amount ?? 0), 0);
}

export function getMonthlyAverage(
  records: SalaryRecord[],
  year: number
): number {
  const monthsWithIncome = new Set(
    records
      .filter(
        (record) =>
          record.year === year &&
          record.status === "received" &&
          (record.amount ?? 0) > 0
      )
      .map((record) => record.month)
  );

  if (monthsWithIncome.size === 0) return 0;

  return getAnnualTotal(records, year) / monthsWithIncome.size;
}

export function getSalaryVariation(
  current: number,
  previous: number
): number | null {
  if (previous === 0) return null;

  return ((current - previous) / previous) * 100;
}

export function getSourcesByCategory(sources: SalarySource[]) {
  return sources.reduce<Record<string, number>>((result, source) => {
    result[source.category] = (result[source.category] ?? 0) + 1;
    return result;
  }, {});
}

### 3.4. Importação Inicial (`src/components/salary/SalaryImport.tsx`)
```tsx
import { useState } from "react";
import { importSalaryExcel } from "../../services/salaryExcelImport";
import type { SalaryImportResult } from "../../types/salary";

interface Props {
  onImport: (data: SalaryImportResult) => Promise<void>;
}

export default function SalaryImport({ onImport }: Props) {
  const [preview, setPreview] = useState<SalaryImportResult null |>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  async function handleFile(file?: File) {
    if (!file) return;

    setLoading(true);
    setError("");
    setPreview(null);

    try {
      const result = await importSalaryExcel(file);
      setPreview(result);
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Erro ao ler a planilha."
      );
    } finally {
      setLoading(false);
    }
  }

  async function confirmImport() {
    if (!preview) return;

    setLoading(true);

    try {
      await onImport(preview);
      setPreview(null);
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Erro ao salvar os dados."
      );
    } finally {
      setLoading(false);
    }
  }

  return (
    <section className="space-y-4 rounded-xl border p-4">
      <h2 className="text-xl font-bold">Importar salários do Excel</h2>

      <input
        type="file"
        accept=".xlsx,.xls"
        onChange={(event) => handleFile(event.target.files?.[0])}
      />

      {loading && <p>Processando planilha...</p>}

      {error && <p className="text-red-500">{error}</p>}

      {preview && (
        <div className="space-y-3">
          <p>
            Ano identificado: <strong>{preview.year}</strong>
          </p>

          <p>
            Fontes encontradas: <strong>{preview.sources.length}</strong>
          </p>

          <p>
            Lançamentos mensais: <strong>{preview.records.length}</strong>
          </p>

          {preview.warnings.map((warning, index) => (
            <p key={index} className="text-amber-600">
              {warning}
            </p>
          ))}

          <button
            type="button"
            disabled={loading || preview.errors.length > 0}
            onClick={confirmImport}
            className="rounded-lg bg-blue-600 px-4 py-2 text-white"
          >
            Confirmar importação
          </button>
        </div>
      )}
    </section>
  );
}
